#include <Arduino.h>
#include <WiFi.h>
#include <WebServer.h>
#include <FirebaseESP32.h>
#include "config.h"
#include <DHT.h>

// =======================================
// CONFIGURACIÓN DEL DEVICE ID
// =======================================
const String DEVICE_ID = "device_A4CB2F124B00";

// =======================================
// ASIGNACIÓN DE PINES
// =======================================
#define MQ2_PIN 34       // Cable Blanco (AO del sensor MQ)
#define PM25_PIN 35      // Cable Azul (VO del adaptador PM2.5)
#define PM25_LED_PIN 18  // Cable Morado (LED del adaptador PM2.5)
#define DHT_PIN 23       // Pin de datos del DHT11
#define DHT_TYPE DHT11

DHT dht(DHT_PIN, DHT_TYPE);
WebServer server(80);

FirebaseData fbData;
FirebaseAuth auth;
FirebaseConfig config;

int valorCO = 0;
int valorPM25 = 0;
int valorHumedad = 0;
float temperatura = 0.0;

bool activo = true;
unsigned long ultimoEnvio = 0;
const unsigned long INTERVALO_ENVIO = 5000; // 5 segundos

// =======================================
// Función de lectura precisa para PM2.5 (GP2Y1010AU0F / OPEN-SMART)
// =======================================
int leerSensorPM25() {
  digitalWrite(PM25_LED_PIN, LOW); // Enciende el LED infrarrojo interno
  delayMicroseconds(280);
  
  int lectura = analogRead(PM25_PIN); // Muestreo en el pico de emisión
  
  delayMicroseconds(40);
  digitalWrite(PM25_LED_PIN, HIGH); // Apaga el LED infrarrojo
  delayMicroseconds(9680);          // Completa el ciclo de 10ms recomendado
  
  return lectura;
}

// =======================================
// Página Web Local
// =======================================
void handleRoot() {
  String html = "<html><head><meta charset='UTF-8'></head><body style='font-family:sans-serif;'>";
  html += "<h2>🌎 Monitor Ambiental (" + DEVICE_ID + ")</h2>";
  html += "<p><strong>Estado:</strong> " + String(activo ? "Activo ✅" : "Detenido ⛔") + "</p>";
  html += "<p>CO (MQ): " + String(valorCO) + "</p>";
  html += "<p>PM2.5: " + String(valorPM25) + "</p>";
  html += "<p>Humedad: " + String(valorHumedad) + "%</p>";
  html += "<p>Temperatura: " + String(temperatura, 2) + "°C</p>";
  html += "<p><a href='/stop'>🛑 Detener</a> | <a href='/start'>▶ Reanudar</a></p>";
  html += "</body></html>";

  server.send(200, "text/html", html);
}

void handleStop() {
  activo = false;
  server.send(200, "text/html", "<h2>🛑 Envío detenido</h2><a href='/'>Volver</a>");
}

void handleStart() {
  activo = true;
  server.send(200, "text/html", "<h2>▶ Reanudado</h2><a href='/'>Volver</a>");
}

// =======================================
// Enviar estado actual a Firebase RTDB
// =======================================
void actualizarEstadoRTDB() {
  FirebaseJson data;
  data.set("CO", valorCO);
  data.set("PM2_5", valorPM25);
  data.set("humedad", valorHumedad);
  data.set("temperatura", temperatura);
  data.set("last_update_server/.sv", "timestamp");

  String path = "/dispositivos/" + DEVICE_ID;
  if (!Firebase.setJSON(fbData, path, data)) {
    Serial.println("❌ Error RTDB: " + fbData.errorReason());
  }
}

// =======================================
// Guardar histórico en Firebase RTDB
// =======================================
void guardarLecturaHistorica() {
  FirebaseJson data;
  data.set("CO", valorCO);
  data.set("PM2_5", valorPM25);
  data.set("humedad", valorHumedad);
  data.set("temperatura", temperatura);
  data.set("timestamp_server/.sv", "timestamp");

  String path = "/dispositivos/" + DEVICE_ID + "/historial";
  if (!Firebase.pushJSON(fbData, path, data)) {
    Serial.println("❌ Error Historial: " + fbData.errorReason());
  }
}

// =======================================
// SETUP
// =======================================
void setup() {
  Serial.begin(115200);

  pinMode(PM25_LED_PIN, OUTPUT);
  digitalWrite(PM25_LED_PIN, HIGH); // Apagado por defecto

  dht.begin();

  Serial.print("Conectando WiFi...");
  WiFi.begin(WIFI_SSID, WIFI_PASS);

  while (WiFi.status() != WL_CONNECTED) {
    Serial.print(".");
    delay(400);
  }
  Serial.println("\nWiFi conectado. IP: " + WiFi.localIP().toString());

  config.database_url = FIREBASE_HOST;
  config.signer.tokens.legacy_token = FIREBASE_AUTH;

  Firebase.begin(&config, &auth);
  Firebase.reconnectWiFi(true);

  server.on("/", handleRoot);
  server.on("/stop", handleStop);
  server.on("/start", handleStart);
  server.begin();

  Serial.println("Sistema iniciado correctamente.");
}

// =======================================
// LOOP
// =======================================
void loop() {
  server.handleClient();

  // Control manual por consola Serial
  if (Serial.available()) {
    char comando = Serial.read();
    if (comando == 'P' || comando == 'p') {
      activo = false;
      Serial.println("⛔ ESP32 pausado (P)");
    } else if (comando == 'R' || comando == 'r') {
      activo = true;
      Serial.println("▶ ESP32 reanudado (R)");
    }
  }

  // Temporización no bloqueante
  if (activo && (millis() - ultimoEnvio >= INTERVALO_ENVIO)) {
    ultimoEnvio = millis();

    // 1. Lecturas analógicas
    valorCO = analogRead(MQ2_PIN);
    valorPM25 = leerSensorPM25();

    // 2. Lecturas DHT
    float h = dht.readHumidity();
    float t = dht.readTemperature();

    if (!isnan(h) && !isnan(t)) {
      valorHumedad = (int)h;
      temperatura = t;
    }

    // 3. Envío a la base de datos
    actualizarEstadoRTDB();
    guardarLecturaHistorica();

    // 4. Salida Serial
    Serial.printf("📊 CO=%d | PM2.5=%d | Hum=%d%% | Temp=%.2f°C\n",
                  valorCO, valorPM25, valorHumedad, temperatura);
  }
}
